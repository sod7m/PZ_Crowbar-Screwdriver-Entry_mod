module Base
{
    imports
    {
        Base
    }

    /*
     * Home canning recipes. All outputs are vanilla items so disabling the
     * feature mid-save cannot leave dangling references. Recipes are gated
     * to Cooking 3 to keep the canning loop from trivialising the food
     * economy. The CSR_HomeCanning.lua hook hides these recipes from the
     * craft window when EnableHomeCanning is false or when an external
     * canning recipe mod is detected, so duplicate entries are avoided.
     */

    craftRecipe CSR_MakeCannedEvaporatedMilk
    {
        timedAction = MixingBucket,
        time = 60,
        Tags = InHandCraft,
        category = CommonSenseReborn,
        SkillRequired = Cooking:3,
        inputs
        {
            item 1 [Base.TinCanEmpty] mode:destroy flags[InheritCondition;ItemCount],
            item 1 [Base.JarLid] mode:destroy,
            -fluid 1.0 [CowMilk] mode:mixture,
        }
        outputs
        {
            item 1 Base.CannedMilk,
        }
    }
